Run this Python 3.12+ snippet from the repository root to reconstruct and verify
all retained evidence. It does not rerun numerical generation. The output path is
new, so historical artifacts are not overwritten. Fresh-execution commands are in
0028-standardization-reproduction.txt; source archive identities are also retained.

```python
import hashlib
import io
import json
import subprocess
import tarfile
import tempfile
import zipfile
from pathlib import Path

sha = lambda data: hashlib.sha256(data).hexdigest()
research = Path("docs/research")
meta = json.loads((research / "0028-standardization-spec-artifacts.json").read_text())
parts = []
for item in meta["parts"]:
    data = (research / item["path"]).read_bytes()
    assert len(data) == item["bytes"] and sha(data) == item["sha256"]
    parts.append(data)
archive = b"".join(parts)
assert sha(archive) == meta["archive_sha256"]
out = Path(tempfile.mkdtemp(prefix="standardization-retained-"))
with zipfile.ZipFile(io.BytesIO(archive)) as zipped:
    zipped.extractall(out)
for destination, source in json.loads((out / "duplicate-paths.json").read_text()).items():
    target = out / destination
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes((out / source).read_bytes())
for item in json.loads((out / "SOURCE_ARCHIVES.json").read_text()).values():
    data = subprocess.check_output(item["git_command"])
    assert len(data) == item["bytes"] and sha(data) == item["sha256"]
    (out / item["archive"]).write_bytes(data)
    with tarfile.open(fileobj=io.BytesIO(data)) as source_archive:
        source_archive.extractall(out / item["extracted_directory"], filter="data")
raw = (out / "MANIFEST.json").read_bytes()
assert sha(raw) == meta["original_manifest_sha256"]
for name, item in json.loads(raw).items():
    data = (out / name).read_bytes()
    assert len(data) == item["bytes"] and sha(data) == item["sha256"], name
print(out)
```

The five final successful programs are common.py, fit.py, recipes.py,
