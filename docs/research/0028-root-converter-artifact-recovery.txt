Recover the retained evidence from the repository root with Python 3.12 or later.
Use a fresh output directory; the snippet verifies every part before extracting.
This reconstructs archived data, not a new numerical verification run. Fresh
execution instructions are in 0028-root-converter-reproduction.txt.

```python
import hashlib
import io
import json
import subprocess
import tarfile
import tempfile
import zipfile
from pathlib import Path

sha = lambda data: hashlib.sha256(data).hexdigest()
research = Path("docs/research")
meta = json.loads((research / "0028-root-converter-spec-artifacts.json").read_text())
chunks = []
for item in meta["parts"]:
    data = (research / item["path"]).read_bytes()
    assert len(data) == item["bytes"] and sha(data) == item["sha256"]
    chunks.append(data)
archive = b"".join(chunks)
assert sha(archive) == meta["archive_sha256"]
out = Path(tempfile.mkdtemp(prefix="root-converter-retained-"))
with zipfile.ZipFile(io.BytesIO(archive)) as zipped:
    zipped.extractall(out)
for destination, source in json.loads((out / "duplicate-paths.json").read_text()).items():
    target = out / destination
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes((out / source).read_bytes())
baseline = subprocess.check_output(["git", "archive", meta["baseline"], "src"])
(out / "baseline-source.tar").write_bytes(baseline)
with tarfile.open(fileobj=io.BytesIO(baseline)) as source_archive:
    source_archive.extractall(out / "baseline", filter="data")
for folder, expected in [(out, meta["original_manifest_sha256"]),
                         (out / "publication", meta["publication_manifest_sha256"])]:
    raw_manifest = (folder / "MANIFEST.json").read_bytes()
    assert sha(raw_manifest) == expected
    for name, expected_hash in json.loads(raw_manifest).items():
        assert sha((folder / name).read_bytes()) == expected_hash, name
print(out)
```

The original and publication manifests cover 611 and 225 files respectively,
excluding themselves and generated Python bytecode. The archive retains 680
logical files, including both manifests, and reconstructs 158 baseline source/
archive files from Git. Duplicate-path entries restore 365 byte-identical copies.
The original harness input error, intermediate verifier attempts, and the
publication-only import-spacing correction remain independently visible.
